    public partial class Form1 : Form
    {
        private string connectionString = "Host=localhost;Port=5432;Username=postgres;Password=12345;Database=estoque;";

        public Form1()
        {
            InitializeComponent();
        }

        private void Form1_Load(object sender, EventArgs e)
        {
            CarregarProdutos();
        }

        private void CarregarProdutos()
        {
            using (var conn = new NpgsqlConnection(connectionString))
            {
                conn.Open();
                string sql = "SELECT * FROM produto ORDER BY id_produto";
                var da = new NpgsqlDataAdapter(sql, conn);
                var dt = new DataTable();
                da.Fill(dt);
                dgvProdutos.DataSource = dt;
            }
        }

        private void btnCadastrar_Click(object sender, EventArgs e)
        {
            using (var conn = new NpgsqlConnection(connectionString))
            {
                conn.Open();
                string sql = "INSERT INTO produto (nome, preco, quantidade) VALUES (@nome, @preco, @quantidade)";
                using (var cmd = new NpgsqlCommand(sql, conn))
                {
                    cmd.Parameters.AddWithValue("nome", txtNome.Text);
                    cmd.Parameters.AddWithValue("preco", decimal.Parse(txtPreco.Text));
                    cmd.Parameters.AddWithValue("quantidade", int.Parse(txtQuantidade.Text));
                    cmd.ExecuteNonQuery();
                }
            }
            CarregarProdutos();
        }

        private void btnAdicionar_Click(object sender, EventArgs e)
        {
            using (var conn = new NpgsqlConnection(connectionString))
            {
                conn.Open();
                string sql = "UPDATE produto SET quantidade = quantidade + @qtd WHERE id_produto = @id";
                using (var cmd = new NpgsqlCommand(sql, conn))
                {
                    cmd.Parameters.AddWithValue("qtd", int.Parse(txtQuantidade.Text));
                    cmd.Parameters.AddWithValue("id", int.Parse(txtId.Text));
                    cmd.ExecuteNonQuery();
                }
            }
            CarregarProdutos();
        }

        private void btnRemover_Click(object sender, EventArgs e)
        {
            using (var conn = new NpgsqlConnection(connectionString))
            {
                conn.Open();
                string sql = "UPDATE produto SET quantidade = quantidade - @qtd WHERE id_produto = @id";
                using (var cmd = new NpgsqlCommand(sql, conn))
                {
                    cmd.Parameters.AddWithValue("qtd", int.Parse(txtQuantidade.Text));
                    cmd.Parameters.AddWithValue("id", int.Parse(txtId.Text));
                    cmd.ExecuteNonQuery();
                }
            }
            CarregarProdutos();
        }
    }
